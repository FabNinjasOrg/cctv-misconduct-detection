# Everything the live view needs. Install with:
#
#     pip install -r requirements.txt
#
# No web framework and no Gemini client any more - this is one program that
# opens a camera, not a service with an API.

opencv-python        # decode the RTSP stream, draw the overlay, write snapshots
numpy
pyyaml               # camera.yaml
ultralytics          # YOLO11-pose (people and keypoints) and YOLO11 (objects, for vandalism)
torch                # MoViNet runs on this directly, not through ultralytics
torchvision

# MoViNet-A0 backbone (MIT) for the fight classifier - installed straight from
# GitHub, so git must be available. The checkpoint riding on top of it is our own
# (models/fight_movinet.pt, trained by training/fight_movinet_kaggle.ipynb), but
# on research-only datasets - retrain on your own footage before anything ships.
git+https://github.com/roggerfq/MoViNet-pytorch.git
